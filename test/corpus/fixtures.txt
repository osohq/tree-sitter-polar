test fixture basics {
    has_role(User{"alice"}, "admin", Organization{"acme"});
    has_role(User{"bob"}, "member", Organization{"acme"});
}

test "roles work" {
    setup {
        fixture basics;
        has_role(User{"carol"}, "member", Organization{"acme"});
    }

    assert allow(User{"alice"}, "read", Organization{"acme"});
    assert_not allow(User{"bob"}, "delete", Organization{"acme"});
}
